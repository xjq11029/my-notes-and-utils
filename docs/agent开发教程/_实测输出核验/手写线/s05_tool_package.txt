== 1. create_params_model 生成的 JSON Schema ==
{
  "properties": {
    "filepath": {
      "description": "文件的绝对路径或相对于cwd的路径",
      "title": "Filepath",
      "type": "string"
    },
    "content": {
      "description": "待写入的文件内容",
      "title": "Content",
      "type": "string"
    },
    "encoding": {
      "default": "utf-8",
      "description": "文件编码",
      "title": "Encoding",
      "type": "string"
    }
  },
  "required": [
    "filepath",
    "content"
  ],
  "title": "write_file_params",
  "type": "object"
}

== 2. Tool.schema()（OpenAI function calling 格式）==
{
  "type": "function",
  "function": {
    "name": "write_file",
    "description": "将内容写入文件",
    "parameters": {
      "type": "object",
      "properties": {
        "filepath": {
          "type": "string",
          "description": "文件的绝对路径或相对于cwd的路径"
        },
        "content": {
          "type": "string",
          "description": "待写入的文件内容"
        },
        "encoding": {
          "type": "string",
          "description": "文件编码"
        }
      },
      "required": [
        "filepath",
        "content"
      ]
    }
  }
}
