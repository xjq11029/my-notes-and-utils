## 场景一：用Python编写数据清洗脚本

### 场景描述
某数据分析团队需要处理一批来自多个渠道的原始销售数据，数据格式不统一：CSV中有缺失值和重复行，JSON中有嵌套结构，部分日期格式不一致。需要编写一个Python脚本完成数据加载、清洗、转换和导出，最终输出干净的结构化数据供后续分析使用。

### 涉及知识点
- [Python核心语法速通](./01-Python核心语法速通-导览.md)：列表推导式、字典操作、函数定义、异常处理、文件I/O、标准库csv和json模块

### 协作流程
1. 使用`csv.DictReader`和`json.load()`分别加载CSV和JSON数据源，统一转为字典列表。
2. 用列表推导式过滤空行和重复行：`[row for row in data if row.get('id')]`，配合`set`去重。
3. 用`datetime.strptime`统一日期格式，`try/except`捕获格式异常并记录日志。
4. 用字典映射处理类别编码转换，用`defaultdict`分组聚合。
5. 用上下文管理器`with open()`安全写入清洗后的CSV和JSON文件。
6. 将整个流程封装为`clean_data(input_paths, output_path)`函数，支持命令行参数。

### 关键要点
- 使用生成器`(row for row in reader)`处理大文件，避免全部加载到内存
- 日期处理用`dateutil.parser.parse`兼容更多格式，但需注意性能
- 缺失值填充策略根据业务决定：数值用均值/中位数，类别用众数或"未知"
- 异常处理需记录行号和错误原因，方便回溯

import csv

import random
from datetime import datetime, timedelta
from pathlib import Path

def generate_test_data(output_dir: str = './data'):
    """生成包含脏数据的模拟销售记录"""
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    products = {
        '电子产品': ['手机', '笔记本', '平板', '耳机', '充电器'],
        '服装': ['T恤', '牛仔裤', '连衣裙', '运动鞋', '羽绒服'],
        '食品': ['坚果', '牛奶', '饼干', '巧克力', '茶叶'],
        '家居': ['台灯', '收纳盒', '窗帘', '地毯', '挂钟'],
    }
    channels = ['天猫', '京东', '拼多多', '抖音', '线下']

    records = []
    base_date = datetime(2026, 1, 1)
    for i in range(1, 151):
        category = random.choice(list(products.keys()))
        product = random.choice(products[category])
        price = round(random.uniform(10, 5000), 2)
        qty = random.randint(1, 10)
        order_date = base_date + timedelta(days=random.randint(0, 90))

        record = {
            'id': f'ORD{i:06d}',
            'product': product,
            'category': category,
            'price': price,
            'quantity': qty,
            'amount': round(price * qty, 2),
            'date': order_date.strftime('%Y-%m-%d'),
            'channel': random.choice(channels),
        }
        records.append(record)

    # 注入脏数据：空ID、不同日期格式、缺失金额、缺失类别
    records[10]['id'] = ''
    records[20]['date'] = '2026/03/15'
    records[25]['date'] = '15-03-2026'
    records[30]['amount'] = ''
    records[40]['category'] = ''
    records.append(records[5].copy())  # 重复行

    # 写入 CSV
    csv_path = output_dir / 'sales_2026_q1.csv'
    with open(csv_path, 'w', encoding='utf-8-sig', newline='') as f:
        writer = csv.DictWriter(f, fieldnames=records[0].keys())
        writer.writeheader()
        writer.writerows(records)
    print(f"[INFO] 测试CSV数据已生成: {csv_path} ({len(records)} 行)")

    # 写入 JSON（嵌套结构）
    json_path = output_dir / 'sales_2026_q2.json'
    with open(json_path, 'w', encoding='utf-8') as f:
        json.dump({'source': 'ERP系统', 'records': records[:80]}, f, ensure_ascii=False, indent=2)
    print(f"[INFO] 测试JSON数据已生成: {json_path} (80 行)")

if __name__ == '__main__':
    generate_test_data()

#### 第二步：数据加载模块

```python
"""
clean_and_analyze.py - 数据清洗、分析与可视化主脚本
"""
import csv
import json
import sys
from collections import defaultdict
from datetime import datetime
from pathlib import Path
from typing import List, Dict, Any, Optional


def load_csv_data(filepath: str) -> List[Dict[str, str]]:
    """使用 csv.DictReader 加载 CSV 数据"""
    rows = []
    with open(filepath, 'r', encoding='utf-8') as f:
        reader = csv.DictReader(f)
        for row in reader:
            rows.append(row)
    print(f"[INFO] 加载 CSV: {filepath} ({len(rows)} 行)")
    return rows


def load_json_data(filepath: str) -> List[Dict[str, Any]]:
    """加载 JSON 数据，自动提取 records 字段"""
    with open(filepath, 'r', encoding='utf-8') as f:
        data = json.load(f)
    if isinstance(data, dict):
        records = data.get('records', data.get('data', []))
        if not records:
            records = [data]
    else:
        records = data
    print(f"[INFO] 加载 JSON: {filepath} ({len(records)} 行)")
    return records


def load_all_data(input_paths: List[str]) -> List[Dict[str, Any]]:
    """批量加载多个数据源"""
    all_data = []
    for path in input_paths:
        path = Path(path)
        if path.suffix == '.csv':
            all_data.extend(load_csv_data(str(path)))
        elif path.suffix == '.json':
            all_data.extend(load_json_data(str(path)))
        else:
            print(f"[WARN] 跳过不支持的文件类型: {path}")
    print(f"[INFO] 总计加载 {len(all_data)} 条原始记录")
    return all_data
```

#### 第三步：数据清洗模块

```python
def parse_date_robust(date_str: str) -> Optional[str]:
    """鲁棒日期解析，支持多种格式，统一输出 YYYY-MM-DD"""
    if not date_str or not str(date_str).strip():
        return None
    date_str = str(date_str).strip()

    date_formats = [
        '%Y-%m-%d',           # 2026-01-15
        '%Y/%m/%d',           # 2026/01/15
        '%d-%m-%Y',           # 15-01-2026
        '%d/%m/%Y',           # 15/01/2026
        '%Y%m%d',             # 20260115
        '%Y-%m-%d %H:%M:%S',  # 2026-01-15 10:30:00
    ]
    for fmt in date_formats:
        try:
            return datetime.strptime(date_str, fmt).strftime('%Y-%m-%d')
        except (ValueError, TypeError):
            continue
    print(f"[WARN] 无法解析日期: '{date_str}'")
    return None


def parse_numeric(value: Any) -> float:
    """安全解析数值，无法解析时返回 0"""
    if value is None:
        return 0.0
    try:
        return round(float(str(value).replace(',', '').replace('¥', '')), 2)
    except (ValueError, TypeError):
        return 0.0


def clean_data(records: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
    """数据清洗主流程：去重、过滤空行、标准化字段、统一日期格式"""
    cleaned = []
    seen_ids = set()
    error_log = []

    for idx, row in enumerate(records):
        try:
            # 1. 过滤空行（无 ID）
            order_id = str(row.get('id', row.get('order_id', ''))).strip()
            if not order_id:
                error_log.append(f"行 {idx + 1}: 缺少订单ID，跳过")
                continue

            # 2. 去重
            if order_id in seen_ids:
                error_log.append(f"行 {idx + 1}: 重复订单ID {order_id}，跳过")
                continue
            seen_ids.add(order_id)

            # 3. 标准化字段名并填充缺失值
            clean_row = {
                'id': order_id,
                'product': str(row.get('product', row.get('product_name', ''))).strip(),
                'category': str(row.get('category', row.get('cat', ''))).strip() or '未分类',
                'price': parse_numeric(row.get('price', row.get('unit_price', 0))),
                'quantity': int(row.get('quantity', row.get('qty', 1)) or 1),
                'amount': parse_numeric(row.get('amount', row.get('total', 0))),
                'date': parse_date_robust(str(row.get('date', row.get('order_date', '')))),
                'channel': str(row.get('channel', row.get('source', '未知'))).strip(),
            }

            # 4. 计算缺失金额
            if clean_row['amount'] == 0 and clean_row['price'] > 0:
                clean_row['amount'] = round(clean_row['price'] * clean_row['quantity'], 2)

            cleaned.append(clean_row)

        except Exception as e:
            error_log.append(f"行 {idx + 1}: 处理异常 - {e}")

    print(f"[INFO] 清洗完成: 原始 {len(records)} 条 -> 有效 {len(cleaned)} 条")
    if error_log:
        print(f"[WARN] 跳过 {len(error_log)} 条记录")
        for err in error_log[:5]:
            print(f"  {err}")
    return cleaned
```

#### 第四步：数据分析与聚合模块

```python
def analyze_data(records: List[Dict[str, Any]]) -> Dict[str, Any]:
    """数据分析：按类别、渠道、月份聚合统计"""
    if not records:
        return {'error': '无有效数据'}

    # 按类别聚合
    category_stats = defaultdict(lambda: {'sales': 0.0, 'orders': 0})
    for r in records:
        cat = r['category']
        category_stats[cat]['sales'] += r['amount']
        category_stats[cat]['orders'] += 1

    # 按渠道聚合
    channel_stats = defaultdict(lambda: {'sales': 0.0, 'orders': 0})
    for r in records:
        channel_stats[r['channel']]['sales'] += r['amount']
        channel_stats[r['channel']]['orders'] += 1

    # 按月份聚合
    monthly_stats = defaultdict(lambda: {'sales': 0.0, 'orders': 0})
    for r in records:
        if r['date']:
            month = r['date'][:7]
            monthly_stats[month]['sales'] += r['amount']
            monthly_stats[month]['orders'] += 1

    total_sales = sum(r['amount'] for r in records)
    total_orders = len(records)

    return {
        'total_sales': round(total_sales, 2),
        'total_orders': total_orders,
        'avg_order_value': round(total_sales / total_orders, 2) if total_orders else 0,
        'by_category': dict(sorted(category_stats.items(), key=lambda x: x[1]['sales'], reverse=True)),
        'by_channel': dict(channel_stats),
        'by_month': dict(sorted(monthly_stats.items())),
    }
```

def generate_visualizations(stats: Dict[str, Any], output_dir: str):
    """生成可视化图表：类别销售额柱状图、渠道占比饼图、月度趋势折线图"""
    try:
        import matplotlib
        matplotlib.use('Agg')
        import matplotlib.pyplot as plt

        # 设置中文字体
        plt.rcParams['font.sans-serif'] = ['SimHei', 'Microsoft YaHei', 'DejaVu Sans']
        plt.rcParams['axes.unicode_minus'] = False

        output_dir = Path(output_dir)
        output_dir.mkdir(parents=True, exist_ok=True)

        # 图表1: 各类别销售额柱状图 (Top 10)
        fig, ax = plt.subplots(figsize=(12, 6))
        categories = list(stats['by_category'].keys())[:10]
        sales = [stats['by_category'][c]['sales'] for c in categories]
        bars = ax.bar(categories, sales, color='steelblue', edgecolor='white')
        ax.set_title('各类别销售额排行 (Top 10)', fontsize=16, fontweight='bold')
        ax.set_xlabel('类别', fontsize=12)
        ax.set_ylabel('销售额 (元)', fontsize=12)
        ax.tick_params(axis='x', rotation=45)
        for bar, val in zip(bars, sales):
            ax.text(bar.get_x() + bar.get_width() / 2, bar.get_height() + 50,
                    f'{val:,.0f}', ha='center', va='bottom', fontsize=9)
        plt.tight_layout()
        plt.savefig(output_dir / 'category_sales.png', dpi=150)
        plt.close()
        print(f"[INFO] 图表已保存: {output_dir / 'category_sales.png'}")

        # 图表2: 渠道销售占比饼图
        fig, ax = plt.subplots(figsize=(8, 8))
        channel_data = {k: v['sales'] for k, v in stats['by_channel'].items()}
        colors = ['#ff9999', '#66b3ff', '#99ff99', '#ffcc99', '#c2c2f0']
        ax.pie(channel_data.values(), labels=channel_data.keys(),
               autopct='%1.1f%%', colors=colors[:len(channel_data)],
               startangle=90, explode=[0.02] * len(channel_data))
        ax.set_title('各渠道销售占比', fontsize=16, fontweight='bold')
        plt.tight_layout()
        plt.savefig(output_dir / 'channel_pie.png', dpi=150)
        plt.close()
        print(f"[INFO] 图表已保存: {output_dir / 'channel_pie.png'}")

        # 图表3: 月度销售趋势折线图
        fig, ax = plt.subplots(figsize=(14, 6))
        months = list(stats['by_month'].keys())
        monthly_sales = [stats['by_month'][m]['sales'] for m in months]
        ax.plot(months, monthly_sales, marker='o', linewidth=2, markersize=8, color='#2c7fb8')
        ax.fill_between(range(len(months)), monthly_sales, alpha=0.2, color='#2c7fb8')
        ax.set_title('月度销售趋势', fontsize=16, fontweight='bold')
        ax.set_xlabel('月份', fontsize=12)
        ax.set_ylabel('销售额 (元)', fontsize=12)
        ax.tick_params(axis='x', rotation=45)
        ax.grid(True, alpha=0.3, linestyle='--')
        for i, (m, s) in enumerate(zip(months, monthly_sales)):
            ax.annotate(f'{s:,.0f}', (i, s), textcoords="offset points",
                        xytext=(0, 12), ha='center', fontsize=9)
        plt.tight_layout()
        plt.savefig(output_dir / 'monthly_trend.png', dpi=150)
        plt.close()
        print(f"[INFO] 图表已保存: {output_dir / 'monthly_trend.png'}")

    except ImportError:
        print("[WARN] matplotlib 未安装，跳过图表生成。安装: pip install matplotlib")

#### 第六步：导出与主流程

```python
def export_data(records: List[Dict[str, Any]], stats: Dict[str, Any], output_dir: str):
    """导出清洗后数据和统计报告"""
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    # 导出 CSV
    csv_path = output_dir / 'cleaned_sales_data.csv'
    with open(csv_path, 'w', encoding='utf-8-sig', newline='') as f:
        if records:
            writer = csv.DictWriter(f, fieldnames=records[0].keys())
            writer.writeheader()
            writer.writerows(records)
    print(f"[INFO] 清洗后数据已导出: {csv_path}")

    # 导出 JSON
    json_path = output_dir / 'cleaned_sales_data.json'
    with open(json_path, 'w', encoding='utf-8') as f:
        json.dump(records, f, ensure_ascii=False, indent=2)
    print(f"[INFO] 清洗后数据已导出: {json_path}")

    # 导出统计报告
    report_path = output_dir / 'analysis_report.json'
    with open(report_path, 'w', encoding='utf-8') as f:
        json.dump(stats, f, ensure_ascii=False, indent=2)
    print(f"[INFO] 分析报告已导出: {report_path}")

    # 打印摘要
    print("\n" + "=" * 60)
    print("  销售数据统计分析报告")
    print("=" * 60)
    print(f"  总销售额:    ¥{stats['total_sales']:>12,.2f}")
    print(f"  总订单数:    {stats['total_orders']:>12,}")
    print(f"  平均客单价:  ¥{stats['avg_order_value']:>12,.2f}")
    print(f"  数据类别数:  {len(stats['by_category']):>12}")
    print(f"  销售渠道数:  {len(stats['by_channel']):>12}")
    print("=" * 60)


def main():
    """主流程：加载 -> 清洗 -> 分析 -> 可视化 -> 导出"""
    input_paths = ['./data/sales_2026_q1.csv', './data/sales_2026_q2.json']
    output_dir = './output'

    if len(sys.argv) > 1:
        input_paths = sys.argv[1:-1] if len(sys.argv) > 2 else [sys.argv[1]]
    if len(sys.argv) > 2:
        output_dir = sys.argv[-1]

    print("=" * 60)
    print("  数据清洗与分析流水线")
    print("=" * 60)

    # 步骤1: 加载数据
    print("\n[步骤1] 加载数据...")
    raw_data = load_all_data(input_paths)
    if not raw_data:
        print("[ERROR] 未加载到任何数据")
        sys.exit(1)

    # 步骤2: 清洗数据
    print("\n[步骤2] 清洗数据...")
    cleaned_data = clean_data(raw_data)

    # 步骤3: 分析数据
    print("\n[步骤3] 分析数据...")
    statistics = analyze_data(cleaned_data)

    # 步骤4: 生成可视化
    print("\n[步骤4] 生成可视化图表...")
    generate_visualizations(statistics, output_dir)

    # 步骤5: 导出结果
    print("\n[步骤5] 导出结果...")
    export_data(cleaned_data, statistics, output_dir)

    print(f"\n[SUCCESS] 数据处理完成! 结果保存在: {output_dir}/")


if __name__ == '__main__':
    main()
```
