## 场景一：用LangChain构建RAG问答系统

### 场景描述
某企业内部有大量产品文档、技术手册和FAQ散落在不同系统中，客服人员查找效率低。需要构建一个基于LangChain的RAG问答系统，用户用自然语言提问，系统从文档库中检索相关内容并生成准确回答，支持引用来源。

### 涉及知识点
- [LangChain核心导览](./02-LangChain与AI智能体核心-导览.md)：Chain构建、PromptTemplate、Model I/O
- [RAG实战导览](./03-RAG检索增强生成实战-导览.md)：文档加载、文本分块、向量化、检索策略、生成回答

### 协作流程
1. 使用LangChain的Document Loaders（DirectoryLoader、PyPDFLoader）加载多格式文档。
2. 用RecursiveCharacterTextSplitter按语义边界分块，chunk_size=800，chunk_overlap=150。
3. 用OpenAI Embeddings或本地模型将文档块向量化，存入Chroma向量数据库。
4. 构建RetrievalQA Chain：用户问题经embedding转为向量，在Chroma中检索Top-K相似文档块。
5. 将检索到的文档块作为上下文，与用户问题一起填入PromptTemplate，调用LLM生成回答。
6. 用LangSmith追踪整个检索和生成流程，监控延迟和准确率。

### 关键要点
- 文档分块策略直接影响检索质量，需根据文档类型调优
- 检索结果需设置相似度阈值过滤低质量文档
- Prompt中要明确要求LLM基于上下文回答，不知道时说"不知道"
- 引入RAGAS评估体系监控Faithfulness和Answer Relevance

```python
# document_loader.py - 文档加载器
import os
from pathlib import Path
from typing import List
from langchain_community.document_loaders import (
    DirectoryLoader,
    PyPDFLoader,
    TextLoader,
    UnstructuredMarkdownLoader,
)
from langchain_core.documents import Document
from config import DOCUMENTS_DIRECTORY


def load_documents(docs_dir: str = DOCUMENTS_DIRECTORY) -> List[Document]:
    """
    加载目录下多种格式文档：PDF、TXT、MD
    根据文件后缀自动选择加载器
    """
    docs_dir = Path(docs_dir)
    if not docs_dir.exists():
        raise FileNotFoundError(f"文档目录不存在: {docs_dir}")

    # 加载 PDF
    pdf_loader = DirectoryLoader(
        docs_dir,
        glob="**/*.pdf",
        loader_cls=PyPDFLoader,
        show_progress=True,
    )
    pdf_docs = pdf_loader.load()
    print(f"[INFO] 加载 PDF 文档: {len(pdf_docs)} 页")

    # 加载 TXT
    txt_loader = DirectoryLoader(
        docs_dir,
        glob="**/*.txt",
        loader_cls=TextLoader,
        loader_kwargs={"encoding": "utf-8"},
    )
    txt_docs = txt_loader.load()
    print(f"[INFO] 加载 TXT 文档: {len(txt_docs)} 份")

    # 加载 Markdown
    md_loader = DirectoryLoader(
        docs_dir,
        glob="**/*.md",
        loader_cls=UnstructuredMarkdownLoader,
    )
    md_docs = md_loader.load()
    print(f"[INFO] 加载 Markdown 文档: {len(md_docs)} 份")

    all_docs = pdf_docs + txt_docs + md_docs
    print(f"[INFO] 总计加载文档: {len(all_docs)} 块")
    return all_docs
```

```python
# vector_store.py - 构建向量库
from typing import List
from langchain.text_splitter import RecursiveCharacterTextSplitter
# 注意：langchain 0.2.x 中 RecursiveCharacterTextSplitter 已迁移至 langchain_text_splitters
# 0.2.x 版本请使用: from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_core.documents import Document
from langchain_openai import OpenAIEmbeddings
from langchain_chroma import Chroma
from config import (
    CHUNK_SIZE,
    CHUNK_OVERLAP,
    PERSIST_DIRECTORY,
    OPENAI_API_KEY,
    OPENAI_BASE_URL,
)


def split_documents(documents: List[Document]) -> List[Document]:
    """按语义边界递归分块，设置合理的重叠减少语义割裂"""
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=CHUNK_SIZE,
        chunk_overlap=CHUNK_OVERLAP,
        length_function=len,
        add_start_index=True,  # 记录在原文档中的位置，方便溯源
        separators=["\n\n", "\n", "。", "，", " ", ""],
    )
    chunks = text_splitter.split_documents(documents)
    print(f"[INFO] 分块: {len(documents)} 文档 -> {len(chunks)} 块")
    return chunks


def build_vector_store(chunks: List[Document]) -> Chroma:
    """构建并持久化向量存储"""
    embeddings = OpenAIEmbeddings(
        api_key=OPENAI_API_KEY,
        base_url=OPENAI_BASE_URL,
    )

    vector_db = Chroma.from_documents(
        documents=chunks,
        embedding=embeddings,
        persist_directory=PERSIST_DIRECTORY,
        collection_name="knowledge_base",
    )
    vector_db.persist()
    print(f"[INFO] 向量库已构建并持久化到: {PERSIST_DIRECTORY}")
    return vector_db


def load_vector_store() -> Chroma:
    """加载已构建的向量库"""
    embeddings = OpenAIEmbeddings(
        api_key=OPENAI_API_KEY,
        base_url=OPENAI_BASE_URL,
    )
    vector_db = Chroma(
        persist_directory=PERSIST_DIRECTORY,
        embedding=embeddings,
        collection_name="knowledge_base",
    )
    print(f"[INFO] 向量库加载完成，集合大小: {vector_db._collection.count()}")
    return vector_db


if __name__ == "__main__":
    from document_loader import load_documents

    print("开始构建知识库...")
    docs = load_documents()
    chunks = split_documents(docs)
    build_vector_store(chunks)
    print("知识库构建完成!")
```

#### 第五步：构建问答链

```python
# rag_chain.py - 构建检索问答链
from langchain_openai import ChatOpenAI
from langchain_chroma import Chroma
from langchain.chains import RetrievalQA
from langchain.prompts import PromptTemplate
from config import (
    OPENAI_API_KEY,
    OPENAI_BASE_URL,
    TOP_K,
    SCORE_THRESHOLD,
)


def create_rag_chain(vector_db: Chroma) -> RetrievalQA:
    """
    构建 RAG 问答链：
    1. 用户问题 -> embedding -> 向量检索
    2. 检索到的文档 + 用户问题 -> Prompt 模板
    3. 调用 LLM 生成回答
    """
    # 配置检索器：相似度阈值过滤，只保留 TOP_K
    # 注意：必须指定 search_type="similarity_score_threshold"，否则 score_threshold 不生效
    retriever = vector_db.as_retriever(
        search_type="similarity_score_threshold",
        search_kwargs={
            "k": TOP_K,
            "score_threshold": SCORE_THRESHOLD,
        }
    )

    # Prompt 模板：明确要求基于上下文回答
    prompt_template = """你是一个专业的知识库问答助手，请根据提供的上下文信息回答用户的问题。
如果上下文信息中找不到答案，请直接说"抱歉，我在知识库中找不到相关信息，无法回答这个问题"，不要编造答案。

上下文信息：
{context}

问题：{question}

请给出准确回答，并引用来源文档位置（如果有）："""

    prompt = PromptTemplate(
        template=prompt_template,
        input_variables=["context", "question"],
    )

    # 初始化 LLM
    llm = ChatOpenAI(
        model="gpt-3.5-turbo",
        temperature=0.1,  # 低温度，回答更稳定准确
        api_key=OPENAI_API_KEY,
        base_url=OPENAI_BASE_URL,
    )

    # 构建检索问答链
    qa_chain = RetrievalQA.from_chain_type(
        llm=llm,
        chain_type="stuff",  # 把所有检索到的文档都塞进上下文
        retriever=retriever,
        chain_type_kwargs={"prompt": prompt},
        return_source_documents=True,  # 返回检索到的源文档用于溯源
    )

    print("[INFO] RAG 问答链构建完成")
    return qa_chain
```

#### 第六步：问答交互主程序

```python
# main.py - 问答交互入口
from rag_chain import create_rag_chain
from vector_store import load_vector_store


def answer_question(question: str, qa_chain):
    """回答一个问题，并返回结果和来源"""
    result = qa_chain.invoke({"query": question})
    answer = result["result"]
    sources = result["source_documents"]

    # 整理来源信息
    source_info = []
    for i, doc in enumerate(sources, 1):
        file_path = doc.metadata.get("source", "未知文件")
        page = doc.metadata.get("page", "未知页")
        source_info.append(f"[{i}] {file_path} (页码: {page})")

    return {
        "question": question,
        "answer": answer,
        "sources": source_info,
    }


def interactive_chat():
    """交互式问答"""
    print("=" * 70)
    print("  本地知识库问答系统 (RAG)")
    print("  输入问题提问，输入 'exit' 退出，输入 'sources' 显示上一次来源")
    print("=" * 70)

    # 加载向量库和问答链
    vector_db = load_vector_store()
    qa_chain = create_rag_chain(vector_db)

    last_sources = []
    while True:
        print("\n> ", end="")
        query = input().strip()
        if query.lower() in ("exit", "quit", "q"):
            print("再见!")
            break
        if query.lower() == "sources":
            if last_sources:
                print("\n【引用来源】")
                for src in last_sources:
                    print(f"  - {src}")
            else:
                print("无历史来源")
            continue
        if not query:
            continue

        result = answer_question(query, qa_chain)
        print(f"\n【回答】\n{result['answer']}")
        last_sources = result["sources"]


if __name__ == "__main__":
    interactive_chat()
```

```python
# document_loader.py - 多部门文档加载与分块
from pathlib import Path
from typing import List, Dict
from langchain_community.document_loaders import (
    DirectoryLoader,
    PyPDFLoader,
    TextLoader,
    UnstructuredMarkdownLoader,
)
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_core.documents import Document
from config import DEPARTMENTS, CHUNK_SIZE, CHUNK_OVERLAP


def load_department_documents(dept_key: str) -> List[Document]:
    """
    加载指定部门的文档，支持 PDF、TXT、MD 格式。
    加载时自动为每个文档添加部门来源元数据。
    """
    dept = DEPARTMENTS.get(dept_key)
    if not dept:
        raise ValueError(f"未知部门: {dept_key}")

    docs_dir = Path(dept["dir"])
    if not docs_dir.exists():
        print(f"[WARN] 部门 {dept_key} 的文档目录不存在: {docs_dir}，跳过加载")
        return []

    all_docs = []

    # 加载 PDF
    pdf_loader = DirectoryLoader(
        str(docs_dir), glob="**/*.pdf", loader_cls=PyPDFLoader, show_progress=True
    )
    for doc in pdf_loader.load():
        doc.metadata["department"] = dept_key
        doc.metadata["dept_name"] = dept["name"]
        all_docs.append(doc)

    # 加载 TXT
    txt_loader = DirectoryLoader(
        str(docs_dir), glob="**/*.txt", loader_cls=TextLoader,
        loader_kwargs={"encoding": "utf-8"}
    )
    for doc in txt_loader.load():
        doc.metadata["department"] = dept_key
        doc.metadata["dept_name"] = dept["name"]
        all_docs.append(doc)

    # 加载 Markdown
    md_loader = DirectoryLoader(
        str(docs_dir), glob="**/*.md", loader_cls=UnstructuredMarkdownLoader
    )
    for doc in md_loader.load():
        doc.metadata["department"] = dept_key
        doc.metadata["dept_name"] = dept["name"]
        all_docs.append(doc)

    print(f"[INFO] 部门 [{dept_key}] 加载文档: {len(all_docs)} 块")
    return all_docs


def split_documents(documents: List[Document]) -> List[Document]:
    """
    按语义边界递归分块，中英文混合场景使用适配的分隔符。
    """
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=CHUNK_SIZE,
        chunk_overlap=CHUNK_OVERLAP,
        length_function=len,
        add_start_index=True,
        separators=["\n\n", "\n", "。", "！", "？", "，", " ", ""],
    )
    chunks = text_splitter.split_documents(documents)
    print(f"[INFO] 分块完成: {len(documents)} 文档 -> {len(chunks)} 块")
    return chunks


def load_all_departments() -> Dict[str, List[Document]]:
    """加载所有部门的文档并按部门分组返回"""
    all_dept_docs = {}
    for dept_key in DEPARTMENTS:
        docs = load_department_documents(dept_key)
        if docs:
            all_dept_docs[dept_key] = split_documents(docs)
    total = sum(len(docs) for docs in all_dept_docs.values())
    print(f"[INFO] 全部部门加载完成，总计 {total} 个文档块")
    return all_dept_docs
```

```python
# vector_store.py - 多知识库向量存储管理
from typing import List, Dict, Optional
from langchain_core.documents import Document
from langchain_openai import OpenAIEmbeddings
from langchain_chroma import Chroma
from config import (
    DEPARTMENTS, PERSIST_DIRECTORY,
    OPENAI_API_KEY, OPENAI_BASE_URL,
)


def build_all_vector_stores(dept_docs: Dict[str, List[Document]]) -> Dict[str, Chroma]:
    """
    为每个部门构建独立的向量库，存储到不同的 collection。
    """
    embeddings = OpenAIEmbeddings(
        api_key=OPENAI_API_KEY,
        base_url=OPENAI_BASE_URL,
    )

    vector_stores = {}
    for dept_key, docs in dept_docs.items():
        if not docs:
            continue
        collection_name = DEPARTMENTS[dept_key]["collection"]
        vector_db = Chroma.from_documents(
            documents=docs,
            embedding=embeddings,
            persist_directory=PERSIST_DIRECTORY,
            collection_name=collection_name,
        )
        vector_db.persist()
        vector_stores[dept_key] = vector_db
        print(f"[INFO] 向量库 [{dept_key}] 已构建: collection={collection_name}, "
              f"文档数={len(docs)}")

    return vector_stores


def load_all_vector_stores() -> Dict[str, Chroma]:
    """
    加载所有已构建的部门向量库。
    """
    embeddings = OpenAIEmbeddings(
        api_key=OPENAI_API_KEY,
        base_url=OPENAI_BASE_URL,
    )

    vector_stores = {}
    for dept_key, dept_info in DEPARTMENTS.items():
        try:
            vector_db = Chroma(
                persist_directory=PERSIST_DIRECTORY,
                embedding=embeddings,
                collection_name=dept_info["collection"],
            )
            count = vector_db._collection.count()
            if count > 0:
                vector_stores[dept_key] = vector_db
                print(f"[INFO] 向量库 [{dept_key}] 已加载: {count} 条记录")
        except Exception as e:
            print(f"[WARN] 向量库 [{dept_key}] 加载失败: {e}")

    return vector_stores
```

```python
# rag_chain.py - 多知识库检索问答链 + 智能路由
from typing import List, Dict, Any, Optional
from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_chroma import Chroma
from langchain.chains import RetrievalQA
from langchain.prompts import PromptTemplate
from langchain_core.documents import Document
from config import (
    DEPARTMENTS, OPENAI_API_KEY, OPENAI_BASE_URL,
    TOP_K, SCORE_THRESHOLD,
)


def classify_department(question: str) -> List[str]:
    """
    根据问题关键词自动路由到对应部门知识库。
    返回匹配到的部门列表，按优先级排序。
    """
    # 关键词映射（简化版，生产环境建议使用语义分类模型）
    keyword_map = {
        "hr":       ["请假", "考勤", "薪资", "福利", "社保", "入职", "离职", "HR", "人事"],
        "it":       ["VPN", "密码", "电脑", "打印机", "网络", "邮箱", "IT", "软件安装"],
        "product":  ["功能", "版本", "API", "接口", "SDK", "部署", "配置", "文档"],
    }

    matched = []
    for dept, keywords in keyword_map.items():
        for kw in keywords:
            if kw in question:
                matched.append(dept)
                break

    # 如果没有匹配到任何部门，返回所有部门（全库检索）
    return matched if matched else list(DEPARTMENTS.keys())


def create_rag_chain(
    vector_stores: Dict[str, Chroma],
    target_depts: Optional[List[str]] = None,
) -> RetrievalQA:
    """
    创建 RAG 问答链，支持指定部门知识库检索。
    当 target_depts 为空时，检索所有部门知识库。
    """
    if target_depts is None:
        target_depts = list(vector_stores.keys())

    # 合并多个向量库的检索器
    # 注意：Chroma 暂不支持原生多 collection 联合检索，
    # 这里采用简化方案：依次检索每个知识库并合并结果
    all_docs = []
    for dept_key in target_depts:
        if dept_key not in vector_stores:
            continue
        retriever = vector_stores[dept_key].as_retriever(
            search_type="similarity_score_threshold",
            search_kwargs={"k": TOP_K, "score_threshold": SCORE_THRESHOLD},
        )
        # 这里需要传入 query，但 RetrievalQA 链会自动处理
        # 因此我们使用 MultiRetriever 或自定义方式合并

    # 简化方案：使用第一个匹配到的知识库进行检索
    # 生产环境建议使用 EnsembleRetriever 或自定义合并逻辑
    primary_dept = target_depts[0] if target_depts else "hr"
    if primary_dept not in vector_stores:
        # 回退到第一个可用的知识库
        primary_dept = next(iter(vector_stores.keys()))

    retriever = vector_stores[primary_dept].as_retriever(
        search_type="similarity_score_threshold",
        search_kwargs={"k": TOP_K, "score_threshold": SCORE_THRESHOLD},
    )

    # Prompt 模板：要求基于上下文回答，并引用来源
    prompt_template = """你是一个专业的企业内部知识库 AI 助手。请根据以下上下文信息回答用户的问题。
如果上下文信息中找不到答案，请直接说"抱歉，我在知识库中找不到相关信息，无法回答这个问题"，不要编造答案。

【当前知识库】{dept_info}

【上下文信息】
{context}

【问题】
{question}

请给出准确回答，并注明信息来源（文件路径和页码）："""

    dept_names = ", ".join(
        DEPARTMENTS[d]["name"] for d in target_depts if d in DEPARTMENTS
    )
    prompt = PromptTemplate(
        template=prompt_template,
        input_variables=["context", "question"],
        partial_variables={"dept_info": dept_names},
    )

    llm = ChatOpenAI(
        model="gpt-3.5-turbo",
        temperature=0.1,
        api_key=OPENAI_API_KEY,
        base_url=OPENAI_BASE_URL,
    )

    qa_chain = RetrievalQA.from_chain_type(
        llm=llm,
        chain_type="stuff",
        retriever=retriever,
        chain_type_kwargs={"prompt": prompt},
        return_source_documents=True,
    )

    print(f"[INFO] RAG 问答链已创建，知识库: {dept_names}")
    return qa_chain


def answer_question(
    question: str,
    vector_stores: Dict[str, Chroma],
) -> Dict[str, Any]:
    """
    回答用户问题：
    1. 自动路由到对应部门知识库
    2. 检索并生成回答
    3. 返回回答和来源引用
    """
    target_depts = classify_department(question)
    print(f"[INFO] 问题路由到知识库: {target_depts}")

    qa_chain = create_rag_chain(vector_stores, target_depts)
    result = qa_chain.invoke({"query": question})

    answer = result["result"]
    sources = result["source_documents"]

    source_info = []
    for i, doc in enumerate(sources, 1):
        file_path = doc.metadata.get("source", "未知文件")
        page = doc.metadata.get("page", "未知页")
        department = doc.metadata.get("dept_name", "未知部门")
        source_info.append({
            "index": i,
            "file": file_path,
            "page": page,
            "department": department,
        })

    return {
        "question": question,
        "answer": answer,
        "departments": [DEPARTMENTS[d]["name"] for d in target_depts if d in DEPARTMENTS],
        "sources": source_info,
    }
```

```python
# main.py - 企业知识库 AI 助手 FastAPI 服务
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel, Field
from typing import List, Optional, Dict, Any
import uvicorn

from vector_store import load_all_vector_stores
from rag_chain import answer_question

# ============================================================
# FastAPI 应用初始化
# ============================================================
app = FastAPI(
    title="企业知识库 AI 助手",
    description="基于 LangChain + RAG 的企业内部知识库问答系统",
    version="1.0.0",
)

# CORS 配置（允许前端跨域访问）
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# ============================================================
# 数据模型
# ============================================================
class QuestionRequest(BaseModel):
    question: str = Field(..., description="用户问题", min_length=1, max_length=2000)
    department: Optional[str] = Field(default=None, description="指定部门知识库（hr/it/product），为空则自动路由")

class SourceInfo(BaseModel):
    index: int
    file: str
    page: str
    department: str

class AnswerResponse(BaseModel):
    question: str
    answer: str
    departments: List[str]
    sources: List[SourceInfo]

class HealthResponse(BaseModel):
    status: str
    vector_stores: Dict[str, int]

# ============================================================
# 全局变量：启动时加载向量库
# ============================================================
vector_stores: Dict[str, Any] = {}

@app.on_event("startup")
async def startup_event():
    """应用启动时加载所有向量库"""
    global vector_stores
    print("正在加载知识库向量存储...")
    vector_stores = load_all_vector_stores()
    if not vector_stores:
        print("[WARN] 未找到任何向量库，请先运行 vector_store.py 构建知识库")
    else:
        print(f"已加载 {len(vector_stores)} 个知识库: {list(vector_stores.keys())}")

# ============================================================
# API 路由
# ============================================================
@app.get("/health", response_model=HealthResponse)
async def health_check():
    """健康检查接口"""
    return HealthResponse(
        status="ok",
        vector_stores={k: v._collection.count() for k, v in vector_stores.items()},
    )

@app.post("/ask", response_model=AnswerResponse)
async def ask(request: QuestionRequest):
    """
    知识库问答接口
    - question: 用户问题
    - department: 可选，指定部门知识库（hr/it/product），为空则自动路由
    """
    if not vector_stores:
        raise HTTPException(
            status_code=503,
            detail="知识库未初始化，请先构建向量存储"
        )

    try:
        result = answer_question(request.question, vector_stores)
        return AnswerResponse(**result)
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"问答处理失败: {str(e)}")

@app.get("/departments")
async def list_departments():
    """列出所有可用的知识库部门"""
    return {
        "departments": [
            {
                "key": k,
                "name": v["name"],
                "doc_count": vector_stores[k]._collection.count() if k in vector_stores else 0,
            }
            for k, v in DEPARTMENTS.items()
        ]
    }

# ============================================================
# 程序入口
# ============================================================
if __name__ == "__main__":
    uvicorn.run(
        "main:app",
        host="0.0.0.0",
        port=8000,
        reload=True,
        log_level="info",
    )
```

### 关键要点
- 多知识库路由需设计清晰的分类逻辑，可通过关键词匹配或语义分类
- Memory策略需平衡上下文完整性和token消耗，推荐ConversationSummaryBufferMemory
- 权限控制可在检索阶段按用户角色过滤文档元数据
- 人工兜底机制是生产环境必备，避免Agent在无法回答时编造信息
