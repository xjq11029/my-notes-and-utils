## 场景一：用Python编写数据清洗脚本

### 场景描述
某数据分析团队需要处理一批来自多个渠道的原始销售数据，数据格式不统一：CSV中有缺失值和重复行，JSON中有嵌套结构，部分日期格式不一致。需要编写一个Python脚本完成数据加载、清洗、转换和导出，最终输出干净的结构化数据供后续分析使用。

### 涉及知识点
- [Python核心语法速通](./01-Python核心语法速通-导览.md)：列表推导式、字典操作、函数定义、异常处理、文件I/O、标准库csv和json模块

### 协作流程
1. 使用`csv.DictReader`和`json.load()`分别加载CSV和JSON数据源，统一转为字典列表。
2. 用列表推导式过滤空行和重复行：`[row for row in data if row.get('id')]`，配合`set`去重。
3. 用`datetime.strptime`统一日期格式，`try/except`捕获格式异常并记录日志。
4. 用字典映射处理类别编码转换，用`defaultdict`分组聚合。
5. 用上下文管理器`with open()`安全写入清洗后的CSV和JSON文件。
6. 将整个流程封装为`clean_data(input_paths, output_path)`函数，支持命令行参数。

### 关键要点
- 使用生成器`(row for row in reader)`处理大文件，避免全部加载到内存
- 日期处理用`dateutil.parser.parse`兼容更多格式，但需注意性能
- 缺失值填充策略根据业务决定：数值用均值/中位数，类别用众数或"未知"
- 异常处理需记录行号和错误原因，方便回溯

```python
"""
clean_and_analyze.py - 数据清洗、分析与可视化主脚本
"""
import csv
import json
import sys
from collections import defaultdict
from datetime import datetime
from pathlib import Path
from typing import List, Dict, Any, Optional


def load_csv_data(filepath: str) -> List[Dict[str, str]]:
    """使用 csv.DictReader 加载 CSV 数据"""
    rows = []
    with open(filepath, 'r', encoding='utf-8') as f:
        reader = csv.DictReader(f)
        for row in reader:
            rows.append(row)
    print(f"[INFO] 加载 CSV: {filepath} ({len(rows)} 行)")
    return rows


def load_json_data(filepath: str) -> List[Dict[str, Any]]:
    """加载 JSON 数据，自动提取 records 字段"""
    with open(filepath, 'r', encoding='utf-8') as f:
        data = json.load(f)
    if isinstance(data, dict):
        records = data.get('records', data.get('data', []))
        if not records:
            records = [data]
    else:
        records = data
    print(f"[INFO] 加载 JSON: {filepath} ({len(records)} 行)")
    return records


def load_all_data(input_paths: List[str]) -> List[Dict[str, Any]]:
    """批量加载多个数据源"""
    all_data = []
    for path in input_paths:
        path = Path(path)
        if path.suffix == '.csv':
            all_data.extend(load_csv_data(str(path)))
        elif path.suffix == '.json':
            all_data.extend(load_json_data(str(path)))
        else:
            print(f"[WARN] 跳过不支持的文件类型: {path}")
    print(f"[INFO] 总计加载 {len(all_data)} 条原始记录")
    return all_data
```

#### 第三步：数据清洗模块

```python
def parse_date_robust(date_str: str) -> Optional[str]:
    """鲁棒日期解析，支持多种格式，统一输出 YYYY-MM-DD"""
    if not date_str or not str(date_str).strip():
        return None
    date_str = str(date_str).strip()

    date_formats = [
        '%Y-%m-%d',           # 2026-01-15
        '%Y/%m/%d',           # 2026/01/15
        '%d-%m-%Y',           # 15-01-2026
        '%d/%m/%Y',           # 15/01/2026
        '%Y%m%d',             # 20260115
        '%Y-%m-%d %H:%M:%S',  # 2026-01-15 10:30:00
    ]
    for fmt in date_formats:
        try:
            return datetime.strptime(date_str, fmt).strftime('%Y-%m-%d')
        except (ValueError, TypeError):
            continue
    print(f"[WARN] 无法解析日期: '{date_str}'")
    return None


def parse_numeric(value: Any) -> float:
    """安全解析数值，无法解析时返回 0"""
    if value is None:
        return 0.0
    try:
        return round(float(str(value).replace(',', '').replace('¥', '')), 2)
    except (ValueError, TypeError):
        return 0.0


def clean_data(records: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
    """数据清洗主流程：去重、过滤空行、标准化字段、统一日期格式"""
    cleaned = []
    seen_ids = set()
    error_log = []

    for idx, row in enumerate(records):
        try:
            # 1. 过滤空行（无 ID）
            order_id = str(row.get('id', row.get('order_id', ''))).strip()
            if not order_id:
                error_log.append(f"行 {idx + 1}: 缺少订单ID，跳过")
                continue

            # 2. 去重
            if order_id in seen_ids:
                error_log.append(f"行 {idx + 1}: 重复订单ID {order_id}，跳过")
                continue
            seen_ids.add(order_id)

            # 3. 标准化字段名并填充缺失值
            clean_row = {
                'id': order_id,
                'product': str(row.get('product', row.get('product_name', ''))).strip(),
                'category': str(row.get('category', row.get('cat', ''))).strip() or '未分类',
                'price': parse_numeric(row.get('price', row.get('unit_price', 0))),
                'quantity': int(row.get('quantity', row.get('qty', 1)) or 1),
                'amount': parse_numeric(row.get('amount', row.get('total', 0))),
                'date': parse_date_robust(str(row.get('date', row.get('order_date', '')))),
                'channel': str(row.get('channel', row.get('source', '未知'))).strip(),
            }

            # 4. 计算缺失金额
            if clean_row['amount'] == 0 and clean_row['price'] > 0:
                clean_row['amount'] = round(clean_row['price'] * clean_row['quantity'], 2)

            cleaned.append(clean_row)

        except Exception as e:
            error_log.append(f"行 {idx + 1}: 处理异常 - {e}")

    print(f"[INFO] 清洗完成: 原始 {len(records)} 条 -> 有效 {len(cleaned)} 条")
    if error_log:
        print(f"[WARN] 跳过 {len(error_log)} 条记录")
        for err in error_log[:5]:
            print(f"  {err}")
    return cleaned
```

```python
def analyze_data(records: List[Dict[str, Any]]) -> Dict[str, Any]:
    """数据分析：按类别、渠道、月份聚合统计"""
    if not records:
        return {'error': '无有效数据'}

    # 按类别聚合
    category_stats = defaultdict(lambda: {'sales': 0.0, 'orders': 0})
    for r in records:
        cat = r['category']
        category_stats[cat]['sales'] += r['amount']
        category_stats[cat]['orders'] += 1

    # 按渠道聚合
    channel_stats = defaultdict(lambda: {'sales': 0.0, 'orders': 0})
    for r in records:
        channel_stats[r['channel']]['sales'] += r['amount']
        channel_stats[r['channel']]['orders'] += 1

    # 按月份聚合
    monthly_stats = defaultdict(lambda: {'sales': 0.0, 'orders': 0})
    for r in records:
        if r['date']:
            month = r['date'][:7]
            monthly_stats[month]['sales'] += r['amount']
            monthly_stats[month]['orders'] += 1

    total_sales = sum(r['amount'] for r in records)
    total_orders = len(records)

    return {
        'total_sales': round(total_sales, 2),
        'total_orders': total_orders,
        'avg_order_value': round(total_sales / total_orders, 2) if total_orders else 0,
        'by_category': dict(sorted(category_stats.items(), key=lambda x: x[1]['sales'], reverse=True)),
        'by_channel': dict(channel_stats),
        'by_month': dict(sorted(monthly_stats.items())),
    }
```

```python
def export_data(records: List[Dict[str, Any]], stats: Dict[str, Any], output_dir: str):
    """导出清洗后数据和统计报告"""
    output_dir = Path(output_dir)

def main():
    """主流程：加载 -> 清洗 -> 分析 -> 可视化 -> 导出"""

        input_paths = sys.argv[1:-1] if len(sys.argv) > 2 else [sys.argv[1]]

    export_data(cleaned_data, statistics, output_dir)

if __name__ == '__main__':
    main()
```

## 场景二：用生成器 + 上下文管理器实现大文件流式处理管道

### 场景描述
运维团队需要每天分析一个约 20GB 的 Nginx 访问日志：统计 Top 10 来源 IP、各状态码占比、按小时分布的错误数，并输出 JSON 报告。要求内存占用恒定（不超过几百 MB）、能容忍少量脏数据、且文件句柄必须可靠释放——不能因为某一行解析失败就中断整个任务。

```mermaid
%%{init: {'themeVariables': {'fontSize': '16px'}}}%%
flowchart LR
    A["逐行读取"] --> B["解析记录"]
    B --> C["过滤与聚合"]
    C --> D["写出报告"]
```

> 上图展示了流式处理管道的四段式结构。每一段都是一个生成器或上下文管理器，数据以"一次一行"的粒度流过整条链路，因此内存占用与文件总大小无关。

### 涉及知识点
- [Python核心语法速通](./01-Python核心语法速通-导览.md)：生成器与 `yield`、`yield from`、上下文管理器、`collections.Counter`/`defaultdict`、`heapq`、异常处理、文件 I/O
- [Python笔面试题集](./02-Python笔面试题集-导览.md)：迭代器协议与惰性求值、生成器只能消费一次、`with` 语句执行流程、坏行容错与异常吞没

### 协作流程
1. 用 `@contextmanager` 封装 `open`，保证无论是否异常文件句柄都会关闭。
2. 用生成器 `read_lines` 逐行读取并剔除空行，内存 O(1)。
3. 用生成器 `parse_records` 做正则解析，坏行直接 `continue` 跳过并计数，不中断管道。
4. 用 `yield from` 把「读取 → 解析」两级生成器串成一条惰性链。
5. 在 `analyze` 中单次遍历完成全部聚合，用 `Counter` 与 `defaultdict` 承载状态。
6. 用 `heapq.nlargest` 取 Top-K，避免对全量数据排序。
7. 用 `with open()` 写出 JSON 报告，并通过上下文管理器统计总耗时。

### 关键要点
- **惰性求值是核心**：整条链路只在 `for` 循环驱动下推进，任何环节都不会预构建完整列表。
- **天然背压**：下游消费慢，上游生成器就自动暂停，无需额外的队列或信号量。
- **生成器只能消费一次**：聚合必须在一次遍历中完成；若需重复遍历，应转 `list` 或改造成可重建的工厂函数。
- **坏数据不应升级为致命错误**：用 `continue` + 计数器，比 `try/except` 包裹整段循环更清晰，也避免异常吞没。
- **编码容错**：`errors="replace"` 可防止个别乱码字节触发 `UnicodeDecodeError` 终止 20GB 的任务。
- **横向扩展**：需要提速时按文件分片交给 `multiprocessing.Pool`，各分片的 `Counter` 支持直接相加合并。

#### 第二步：上下文管理器与生成器管道

```python
"""
stream_pipeline.py - 生成器 + 上下文管理器的流式日志处理管道
"""
import heapq
import json
import re
import sys
import time
from collections import Counter, defaultdict
from contextlib import contextmanager
from pathlib import Path

LOG_PATTERN = re.compile(
    r'(?P<ip>\d+\.\d+\.\d+\.\d+)\s+-\s+-\s+'
    r'\[(?P<ts>[^\]]+)\]\s+'
    r'"(?P<method>\S+)\s+(?P<path>\S+)\s+[^"]*"\s+'
    r'(?P<status>\d{3})\s+(?P<size>\d+)'
)


@contextmanager
def managed_open(path, mode="r", encoding="utf-8", errors="strict"):
    """上下文管理器：保证文件句柄在任何情况下都被关闭"""
    f = open(path, mode, encoding=encoding, errors=errors)
    try:
        yield f
    finally:
        f.close()


@contextmanager
def timer(label: str):
    """上下文管理器：统计代码块耗时"""
    start = time.perf_counter()
    try:
        yield
    finally:
        print(f"[TIMER] {label}: {time.perf_counter() - start:.3f}s")


def read_lines(fh):
    """生成器：逐行读取并剔除空行，内存 O(1)"""
    for line in fh:
        line = line.strip()
        if line:
            yield line


def parse_records(lines, bad_counter):
    """生成器：正则解析，坏行跳过并计数"""
    for line in lines:
        m = LOG_PATTERN.search(line)
        if not m:
            bad_counter["bad_lines"] += 1
            continue
        ts = m.group("ts")
        yield {
            "ip": m.group("ip"),
            "hour": ts[12:14] if len(ts) > 14 else "??",
            "status": int(m.group("status")),
            "size": int(m.group("size")),
        }


def iter_records(path):
    """用 yield from 把两级生成器串成一条惰性链"""
    bad_counter = Counter()
    with managed_open(path, errors="replace") as fh:
        yield from parse_records(read_lines(fh), bad_counter)
    # 管道结束后把坏行计数回传（通过生成器 return 值）
    return bad_counter


def analyze(path: str, top_k: int = 10) -> dict:
    """单次遍历完成全部聚合"""
    ip_counter = Counter()
    hourly_errors = defaultdict(int)
    status_bucket = Counter()
    total = total_bytes = 0
    bad_lines = Counter()

    gen = iter_records(path)
    while True:
        try:
            rec = next(gen)                  # 手动驱动，才能拿到生成器的 return 值
        except StopIteration as stop:        # 管道结束，取回坏行计数
            bad_lines = stop.value or Counter()
            break
        total += 1
        total_bytes += rec["size"]
        ip_counter[rec["ip"]] += 1
        status_bucket[f"{rec['status'] // 100}xx"] += 1
        if rec["status"] >= 500:
            hourly_errors[rec["hour"]] += 1

    return {
        "total_requests": total,
        "total_bytes": total_bytes,
        "top_ips": heapq.nlargest(top_k, ip_counter.items(), key=lambda x: x[1]),
        "status_bucket": dict(status_bucket),
        "5xx_ratio": round(status_bucket["5xx"] / total, 4) if total else 0.0,
        "hourly_errors": dict(sorted(hourly_errors.items())),
        "bad_lines": bad_lines.get("bad_lines", 0),
    }


def main():
    log_path = sys.argv[1] if len(sys.argv) > 1 else "./data/access.log"
    report_path = sys.argv[2] if len(sys.argv) > 2 else "./output/log_report.json"

    with timer("全流程"):
        stats = analyze(log_path)

        out = Path(report_path)
        out.parent.mkdir(parents=True, exist_ok=True)
        with managed_open(out, "w") as f:
            json.dump(stats, f, ensure_ascii=False, indent=2)

    print("=" * 60)
    print("  日志流式分析报告")
    print("=" * 60)
    print(f"  总请求数:   {stats['total_requests']:>12,}")
    print(f"  总流量:     {stats['total_bytes'] / 1024 / 1024:>11.2f} MB")
    print(f"  5xx 占比:   {stats['5xx_ratio']:>12.2%}")
    print(f"  脏数据行数: {stats['bad_lines']:>12,}")
    print(f"  状态码分布: {stats['status_bucket']}")
    print(f"  Top5 IP:    {stats['top_ips'][:5]}")
    print(f"  按小时错误: {stats['hourly_errors']}")
    print(f"\n[SUCCESS] 报告已写出: {report_path}")


if __name__ == "__main__":
    main()
```

**运行方式：**

```bash

# 2. 运行流式处理管道
```

## 场景三：用装饰器 + 元类实现轻量 ORM 与重试框架

### 场景描述
团队内部需要一个极简的数据访问层：模型类在定义时自动完成字段收集与表名注册；字段赋值要做类型校验；查询支持 `filter_xxx(value)` 链式条件；数据库抖动导致的方法失败要能自动重试且重试次数可配置；同时希望对关键操作统计耗时。要求不引入第三方 ORM 依赖，全部用 Python 语言特性实现。

```mermaid

flowchart LR
    A["元类收集字段"] --> B["描述符校验赋值"]
    B --> C["链式条件查询"]
    C --> D["装饰器重试与计时"]
```

> 上图展示了轻量 ORM 的四层语言特性协作：元类负责"类定义期"的登记，描述符负责"赋值期"的校验，`__getattr__` 负责"查询期"的动态方法生成，装饰器负责"调用期"的重试与埋点。

### 涉及知识点
- [Python核心语法速通](./01-Python核心语法速通-导览.md)：元类、描述符与 `__set_name__`、魔术方法、带参数装饰器、`functools.wraps`、异常链、`classmethod`
- [Python笔面试题集](./02-Python笔面试题集-导览.md)：元类与类创建流程、`__getattr__` 动态属性、带参装饰器展开过程、`@classmethod` 与装饰器叠加顺序、重试退避与异常根因保留

### 协作流程
1. 定义 `Field` 描述符：`__set_name__` 自动获取字段名，`__get__`/`__set__` 实现默认值与类型校验。
2. 定义元类 `ModelMeta`：在 `__new__` 中遍历 `namespace` 收集 `Field`，推导表名并写入全局 `REGISTRY`。
3. 定义 `QuerySet`：用 `__getattr__` 动态生成 `filter_<字段>` 方法并返回自身，实现链式调用。
4. 定义带参数装饰器 `retry`：三层嵌套 + `functools.wraps`，指数退避并用 `raise ... from` 保留根因。
5. 定义上下文管理器 `transaction`：模拟事务提交/回滚，用 `__exit__` 返回值决定是否吞掉异常。
6. 在 `Model` 上用 `@classmethod` 暴露 `objects()` 与 `fetch_all()`，注意 `@classmethod` 必须写在最外层。

### 关键要点
- **元类在类定义时生效**：`class User(Model)` 一执行完，`User._table`/`User._fields`/`REGISTRY` 就已就绪，无需手动注册。
- **描述符是"属性级钩子"**：`__set__` 让 `user.age = "abc"` 立即抛 `TypeError`，比事后校验更早暴露问题。
- **`__getattr__` 只在查找失败时触发**：因此动态生成 `filter_xxx` 是安全的，但方法名拼错不会立即报错，需配合 `raise AttributeError`。
- **装饰器叠加顺序**：`@classmethod` 必须在最外层（`@classmethod` 上、`@retry` 下），否则 `cls` 参数传递会错位。
- **异常链不可丢**：重试耗尽后用 `raise RuntimeError(...) from last_exc` 保留最后一次真实故障，便于定位。
- **事务用上下文管理器表达**：`__exit__` 返回 `False` 让异常继续外抛并触发回滚，返回 `True` 则视为已处理。

### 完整可运行代码示例

```python
"""
mini_orm.py - 用元类 + 描述符 + 装饰器实现的轻量 ORM 与重试框架
"""
import functools
import time
from contextlib import contextmanager
from typing import Any, Callable, Dict, List, Optional, Type

REGISTRY: Dict[str, Type["Model"]] = {}


# ---------- 1. 描述符：字段定义 + 类型校验 + 默认值 ----------
class Field:
    def __init__(self, field_type: type = str, default: Any = None,
                 primary_key: bool = False):
        self.field_type = field_type
        self.default = default
        self.primary_key = primary_key
        self.name: Optional[str] = None

    def __set_name__(self, owner: type, name: str) -> None:
        """类创建时由解释器自动回调，拿到字段名"""
        self.name = name

    def __get__(self, instance: Any, owner: type) -> Any:
        if instance is None:                       # 通过类访问时返回描述符本身
            return self
        return instance.__dict__.get(self.name, self.default)

    def __set__(self, instance: Any, value: Any) -> None:
        if value is not None and not isinstance(value, self.field_type):
            raise TypeError(
                f"字段 {self.name} 期望 {self.field_type.__name__}，"
                f"实际收到 {type(value).__name__}"
            )
        instance.__dict__[self.name] = value


# ---------- 2. 带参数装饰器：重试 + 计时 ----------
def retry(times: int = 3, delay: float = 0.1,
          exceptions: tuple = (Exception,)) -> Callable:
    """失败自动重试（指数退避），耗尽后保留根因抛出"""
    def decorator(func: Callable) -> Callable:
        @functools.wraps(func)                     # 保留 __name__ / __doc__
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            last_exc: Optional[Exception] = None
            for attempt in range(1, times + 1):
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    last_exc = e
                    print(f"  [retry] {func.__name__} 第 {attempt}/{times} 次失败: {e}")
                    if attempt < times:
                        time.sleep(delay * attempt)    # 退避
            raise RuntimeError(f"{func.__name__} 重试 {times} 次后仍失败") from last_exc
        return wrapper
    return decorator


def timed(func: Callable) -> Callable:
    """无参装饰器：统计调用耗时"""
    @functools.wraps(func)
    def wrapper(*args: Any, **kwargs: Any) -> Any:
        start = time.perf_counter()
        result = func(*args, **kwargs)
        print(f"  [timed] {func.__name__}: {time.perf_counter() - start:.4f}s")
        return result
    return wrapper


# ---------- 3. 上下文管理器：模拟事务（yield 出撤销回调列表） ----------
@contextmanager
def transaction(label: str = "tx"):
    undo: List[Callable[[], None]] = []
    print(f"  [tx:{label}] BEGIN")
    try:
        yield undo                              # 调用方往里塞撤销动作
    except Exception as e:
        for fn in reversed(undo):               # 逆序执行撤销
            fn()
        print(f"  [tx:{label}] ROLLBACK ({type(e).__name__}: {e})")
        raise                                   # 不吞异常，继续外抛
    else:
        print(f"  [tx:{label}] COMMIT")


# ---------- 4. 链式查询 ----------
class QuerySet:
    def __init__(self, model: Type["Model"]) -> None:
        self.model = model
        self.conditions: Dict[str, Any] = {}

    def __getattr__(self, item: str) -> Callable:
        """动态生成 filter_<字段> 方法，返回自身以支持链式调用"""
        if item.startswith("filter_"):
            field = item[len("filter_"):]
            if field not in self.model._fields:
                raise AttributeError(f"{self.model.__name__} 无字段 {field}")

            def _filter(value: Any) -> "QuerySet":
                self.conditions[field] = value
                return self
            return _filter
        raise AttributeError(item)

    def all(self) -> List["Model"]:
        rows = self.model._rows
        for k, v in self.conditions.items():
            rows = [r for r in rows if getattr(r, k) == v]
        return rows

    def __iter__(self):
        return iter(self.all())

    def __repr__(self) -> str:
        return f"<QuerySet {self.model.__name__} {self.conditions}>"


# ---------- 5. 元类：类创建时收集字段并注册表 ----------
class ModelMeta(type):
    def __new__(mcs, name: str, bases: tuple, namespace: dict) -> type:
        cls = super().__new__(mcs, name, bases, namespace)
        if name != "Model":                        # 跳过基类自身
            cls._fields = {k: v for k, v in namespace.items() if isinstance(v, Field)}
            pk = [k for k, v in cls._fields.items() if v.primary_key]
            cls._pk = pk[0] if pk else None
            cls._table = name.lower() + "s"
            cls._rows: List[Any] = []
            REGISTRY[cls._table] = cls             # 自动注册
            print(f"[META] 注册表 {cls._table} 字段={list(cls._fields)} 主键={cls._pk}")
        return cls


class Model(metaclass=ModelMeta):
    def __init__(self, **kwargs: Any) -> None:
        for fname, field in self._fields.items():
            setattr(self, fname, kwargs.get(fname, field.default))

    @classmethod
    def objects(cls) -> QuerySet:
        return QuerySet(cls)

    @classmethod
    @timed                                         # 注意叠加顺序：classmethod 在最外层
    @retry(times=3, delay=0.02, exceptions=(ConnectionError,))
    def fetch_all(cls) -> List["Model"]:
        """模拟不稳定数据源：前两次抛 ConnectionError"""
        cls._attempt = getattr(cls, "_attempt", 0) + 1
        if cls._attempt < 3:
            raise ConnectionError("模拟数据库抖动")
        return cls._rows

    def __repr__(self) -> str:
        args = ", ".join(f"{k}={getattr(self, k)!r}" for k in self._fields)
        return f"{type(self).__name__}({args})"


class User(Model):
    id = Field(int, primary_key=True)
    name = Field(str, default="匿名")
    age = Field(int, default=0)


def main():
    print("=" * 60)
    print("  轻量 ORM / 重试框架演示")
    print("=" * 60)

    # 元类注册结果
    print("\n[1] 元类注册结果")
    print(f"  表名: {User._table} | 主键: {User._pk} | 注册表: {list(REGISTRY)}")

    # 描述符类型校验
    print("\n[2] 描述符类型校验")
    try:
        User(id=1, name="Alice", age="不是数字")
    except TypeError as e:
        print(f"  按预期抛出: {e}")

    # 填充数据 + 链式查询
    print("\n[3] 链式查询")
    User._rows.extend([
        User(id=1, name="Alice", age=30),
        User(id=2, name="Bob", age=25),
        User(id=3, name="Alice", age=41),
    ])
    print(f"  filter_name('Alice'): {User.objects().filter_name('Alice').all()}")
    print(f"  再叠加 filter_age(30): "
          f"{User.objects().filter_name('Alice').filter_age(30).all()}")

    # 事务上下文管理器
    print("\n[4] 事务上下文管理器")
    with transaction("insert") as undo:
        User._rows.append(User(id=4, name="Carol", age=28))
        undo.append(lambda: User._rows.pop())
    try:
        with transaction("bad-insert") as undo:
            User._rows.append(User(id=5, name="Dave", age=-1))
            undo.append(lambda: User._rows.pop())
            raise ValueError("年龄非法，回滚")
    except ValueError as e:
        print(f"  事务已回滚: {e}")
    print(f"  当前行数: {len(User._rows)}")

    # 重试装饰器
    print("\n[5] 重试装饰器")
    rows = User.fetch_all()
    print(f"  最终取回 {len(rows)} 行")

    print("\n[SUCCESS] 演示结束")


if __name__ == "__main__":
    main()
```
